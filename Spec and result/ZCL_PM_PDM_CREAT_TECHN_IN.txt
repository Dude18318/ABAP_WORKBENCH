CLASS ZCL_PM_PDM_CREAT_TECHN_IN IMPLEMENTATION.


  METHOD add_to_update_buffer.
    READ TABLE mt_update_buffer ASSIGNING FIELD-SYMBOL(<ls_buf>)
      WITH TABLE KEY k1 COMPONENTS guid = is_pdm_techn-guid.
    IF sy-subrc = 0.
      <ls_buf> = is_pdm_techn.
    ELSE.
      APPEND is_pdm_techn TO mt_update_buffer.
    ENDIF.
  ENDMETHOD.


  METHOD build_arbpl_caches.
    DATA: lt_arbpl_list TYPE eamprt_tt_arbpl,
          lv_werks      TYPE werks_d.

    LOOP AT mt_pdm_techn ASSIGNING FIELD-SYMBOL(<ls_pdm>).
      IF <ls_pdm>-req_mat_pl = '1' OR <ls_pdm>-req_mat_sp = '1' OR <ls_pdm>-req_mat_vr = '1'.
        lv_werks = <ls_pdm>-werks.
      ELSEIF <ls_pdm>-req_eq_pl = '1'.
        IF <ls_pdm>-qmnum IS INITIAL.
          DATA(lv_equnr) = <ls_pdm>-equnr .
        ELSE.
          lv_equnr = mo_dao->get_qmih_equnr( <ls_pdm>-qmnum ).
        ENDIF.
        lv_werks = resolve_iwerk_for_eq(  lv_equnr ).
      ELSE.
        CONTINUE.
      ENDIF.
      CHECK lv_werks IS NOT INITIAL.

      READ TABLE mt_izd_cache INTO DATA(ls_izd) WITH KEY guid_dse = <ls_pdm>-guid.
      CHECK sy-subrc = 0.
      CLEAR lt_arbpl_list.
      LOOP AT ls_izd-tab_tech INTO DATA(ls_tech).        "#EC CI_NESTED
        LOOP AT ls_tech-tab_oper INTO DATA(ls_oper).     "#EC CI_NESTED
          lt_arbpl_list = VALUE #(  BASE  lt_arbpl_list  ( sign = 'I' option = 'EQ' low = ls_oper-arbpl ) ).
        ENDLOOP.
      ENDLOOP.
      CHECK lt_arbpl_list IS NOT INITIAL.

      READ TABLE mt_werks_cache ASSIGNING FIELD-SYMBOL(<ls_werks_cache>) WITH TABLE KEY werks = lv_werks.
      IF sy-subrc <> 0.
        INSERT VALUE #( werks = lv_werks ) INTO TABLE mt_werks_cache ASSIGNING <ls_werks_cache>.
      ENDIF.

      LOOP AT mo_dao->find_work_centers(                 "#EC CI_NESTED
                iv_werks      = lv_werks
                it_arbpl_list = lt_arbpl_list ) INTO DATA(ls_new_cache).

        READ TABLE <ls_werks_cache>-cache
          WITH TABLE KEY k1
            COMPONENTS arbpl_input = ls_new_cache-arbpl_input
          ASSIGNING FIELD-SYMBOL(<ls_upd_cache>).
        IF sy-subrc = 0.
          IF <ls_upd_cache>-arbpl_found IS INITIAL. " Не понятно, может ли быть в одном случае заполнено, а в другом нет
            <ls_upd_cache>-arbpl_found = ls_new_cache-arbpl_found.
          ENDIF.
        ELSE.
          APPEND ls_new_cache TO <ls_werks_cache>-cache. " если нет такой записи, добавим
        ENDIF.
      ENDLOOP.
    ENDLOOP.
  ENDMETHOD.


  METHOD build_notification_text.
    DATA lv_has_technology TYPE flag VALUE abap_false.
    DATA lv_has_kd_only    TYPE flag VALUE abap_true.
    DATA lv_line TYPE string.

    LOOP AT it_pdm_techn ASSIGNING FIELD-SYMBOL(<ls_pdm>)
          WHERE dsenr IS NOT INITIAL.                    "#EC CI_STDSEQ
      IF <ls_pdm>-req_mat_pl = '2' OR <ls_pdm>-req_mat_sp = '2' OR
         <ls_pdm>-req_mat_vr = '2' OR <ls_pdm>-req_eq_pl = '2' OR
         <ls_pdm>-req_eq_sp = '2'.
        lv_has_technology = abap_true.
      ENDIF.
      IF <ls_pdm>-req_mat_pl = '2' OR <ls_pdm>-req_mat_sp = '2' OR
         <ls_pdm>-req_mat_vr = '2' OR <ls_pdm>-req_eq_pl = '2' OR
         <ls_pdm>-req_eq_sp = '2'.
        lv_has_kd_only = abap_false.
      ENDIF.
    ENDLOOP.

    IF lv_has_technology = abap_true.
      rv_text = 'По заявке разработана технология и созданы объекты.'.
    ELSE.
      rv_text = 'По заявке работа выполнена, разработаны КД: '.
    ENDIF.

    LOOP AT it_pdm_techn ASSIGNING <ls_pdm> WHERE dsenr IS NOT INITIAL. "#EC CI_STDSEQ
      IF lv_has_technology = abap_true.
        lv_line = <ls_pdm>-dsenr && ' ' && <ls_pdm>-vertp && ' : ' && cl_abap_char_utilities=>newline .
        IF <ls_pdm>-req_mat_pl = '2' OR <ls_pdm>-req_mat_sp = '2' OR
           <ls_pdm>-req_mat_vr = '2' OR <ls_pdm>-req_eq_pl = '2' OR
           <ls_pdm>-req_eq_sp = '2'.
          IF <ls_pdm>-matnr IS NOT INITIAL.
            lv_line = lv_line && `К материалу ` && |{ <ls_pdm>-matnr ALPHA = OUT }| && ' созданы: ' && cl_abap_char_utilities=>newline  .
          ELSE.
            lv_line = lv_line && `Созданы объекты: ` && cl_abap_char_utilities=>newline  .
          ENDIF.
          IF <ls_pdm>-req_mat_pl = '2'.
            lv_line = lv_line && `- техкарта ` && <ls_pdm>-mat_plnnr && '/' && <ls_pdm>-mat_plnal && '; ' && cl_abap_char_utilities=>newline .
          ENDIF.
          IF <ls_pdm>-req_mat_sp = '2'.
            lv_line = lv_line && `- альт. спецификация ` && <ls_pdm>-mat_stlal  && '; ' && cl_abap_char_utilities=>newline .
          ENDIF.
          IF <ls_pdm>-req_mat_vr = '2'.
            lv_line = lv_line && `- вариант изготовления ` && <ls_pdm>-mat_verid &&  '; ' && cl_abap_char_utilities=>newline .
          ENDIF.
          IF <ls_pdm>-req_eq_pl = '2'.
            lv_line = lv_line && `- ТК ЕО счетчик № ` && <ls_pdm>-eq_plnal &&  '; ' && cl_abap_char_utilities=>newline .
          ENDIF.
          IF <ls_pdm>-req_eq_sp = '2'.
            lv_line = lv_line && `- дополнена спецификация ЕО на материал ` && |{ <ls_pdm>-matnr ALPHA = OUT }|  &&  '; ' && cl_abap_char_utilities=>newline .
          ENDIF.
        ELSE.
          lv_line = lv_line && 'нет технологии'.
        ENDIF.
      ELSE. " только КД
        IF sy-tabix > 1.
          rv_text = rv_text && ', '.
        ENDIF.
        rv_text = rv_text && <ls_pdm>-dsenr && ' ' && <ls_pdm>-vertp.
      ENDIF.
      IF lv_has_technology = abap_true.
        rv_text = rv_text && cl_abap_char_utilities=>newline && lv_line.
      ENDIF.
    ENDLOOP.
  ENDMETHOD.


  METHOD check_deactivation.
    DATA:
      lt_routing_change_ok       TYPE STANDARD TABLE OF string WITH DEFAULT KEY,
      lt_routing_change_err      TYPE STANDARD TABLE OF string WITH DEFAULT KEY,
      lt_mat_bom_maintain_ok     TYPE STANDARD TABLE OF string WITH DEFAULT KEY,
      lt_mat_bom_maintain_err    TYPE STANDARD TABLE OF string WITH DEFAULT KEY,
      lt_prod_vers_db_update_ok  TYPE STANDARD TABLE OF string WITH DEFAULT KEY,
      lt_prod_vers_db_update_err TYPE STANDARD TABLE OF string WITH DEFAULT KEY.
* Выберем все, проверим удалось удалить или нет

*   проверка поочереди
* - метка удаления в «Общих данных» заголовка ТК на ОЗМ;
    DATA(lt_plko_db) =
      mo_dao->get_plko(
        EXPORTING
          iv_get_all = abap_true
          iv_plnty   = CONV #( ms_tvarv-const_4129_pdm_mat_plnty )
          it_keys    = is_deact_obj-t_plko ).
    LOOP AT lt_plko_db INTO DATA(ls_plko_db).
      DATA(lv_routing_change) = |{ ls_plko_db-plnnr } / { ls_plko_db-plnal }|.
      IF ls_plko_db-delkz = abap_true." Удалено
        APPEND lv_routing_change TO lt_routing_change_ok.
      ELSE.   " Еще не удален
        APPEND lv_routing_change TO lt_routing_change_err.
      ENDIF.
    ENDLOOP.

* - установка метки удаления в «Области действия» заголовка спецификации на ОЗМ;
    DATA(lt_stko_db) =
      mo_dao->get_stko(
        EXPORTING
          iv_get_all = abap_true
          iv_stlty   = CONV #( ms_tvarv-const_4129_pdm_mat_stlty )
          it_keys    = is_deact_obj-t_stko ).
    LOOP AT lt_stko_db INTO DATA(ls_stko_db).
      DATA(lv_mat_bom_maintain) = |{ ls_stko_db-stlal }|.
      IF ls_stko_db-loekz = abap_true." Удалено
        APPEND lv_mat_bom_maintain TO lt_mat_bom_maintain_ok.
      ELSE.   " Еще не удален
        APPEND lv_mat_bom_maintain TO lt_mat_bom_maintain_err.
      ENDIF.
    ENDLOOP.

* - установка метки удаления в «Области действия» заголовка спецификации на ОЗМ;
    DATA(lt_mkal_u) = VALUE mkal_tab( ).
    DATA(lt_mkal_keys) =
      VALUE ty_srt_mkal(
        FOR GROUPS ls_mkal_key OF ls_mkal_key_s IN it_prod_versions
        GROUP BY ( matnr = ls_mkal_key_s-matnr
                   werks = ls_mkal_key_s-werks
                   verid = ls_mkal_key_s-verid )
        ASCENDING WITHOUT MEMBERS
          ( matnr = ls_mkal_key-matnr
            werks = ls_mkal_key-werks
            verid = ls_mkal_key-verid ) ).

    DATA(lt_mkal) = mo_dao->get_mkal(
       EXPORTING
         iv_get_all = abap_true
         it_keys    = lt_mkal_keys ).

    LOOP AT it_prod_versions INTO DATA(ls_prod_versions).
      DATA(lv_prod_vers_db_update) = |{ ls_prod_versions-verid }|.
* - заблокировванно для любого использования в варианте изготовления.
      READ TABLE lt_mkal
        WITH TABLE KEY
          matnr = ls_prod_versions-matnr
          werks = ls_prod_versions-werks
          verid = ls_prod_versions-verid
        INTO DATA(ls_mkal).
      IF sy-subrc <> 0 OR ls_mkal-mksp = '1'." Удалено
        APPEND lv_prod_vers_db_update TO lt_prod_vers_db_update_ok.
      ELSE.   " Еще не удален
        APPEND lv_prod_vers_db_update TO lt_prod_vers_db_update_err.
      ENDIF.
    ENDLOOP.

    SORT: lt_routing_change_ok,
          lt_routing_change_err,
          lt_mat_bom_maintain_ok,
          lt_mat_bom_maintain_err,
          lt_prod_vers_db_update_ok,
          lt_prod_vers_db_update_err.
    DELETE ADJACENT DUPLICATES FROM:
      lt_routing_change_ok,
      lt_routing_change_err,
      lt_mat_bom_maintain_ok,
      lt_mat_bom_maintain_err,
      lt_prod_vers_db_update_ok,
      lt_prod_vers_db_update_err.

    IF   lt_mat_bom_maintain_ok IS NOT INITIAL
      OR lt_routing_change_ok IS NOT INITIAL
      OR lt_prod_vers_db_update_ok IS NOT INITIAL.

      _set_ok_deactivation(
        EXPORTING
          iv_stlal       = concat_lines_of(
                              table = lt_mat_bom_maintain_ok
                              sep   = `, ` )
          iv_plnnr_plnal = concat_lines_of(
                              table = lt_routing_change_ok
                              sep   = `, ` )
          iv_verid       = concat_lines_of(
                              table = lt_prod_vers_db_update_ok
                              sep   = `, ` )
        CHANGING
          cv_text        = cv_text ).
    ENDIF.

    IF   lt_mat_bom_maintain_err IS NOT INITIAL
      OR lt_routing_change_err IS NOT INITIAL
      OR lt_prod_vers_db_update_err IS NOT INITIAL.

        ct_text_lines = lt_text_lines ).

    DATA(lv_errors_exist) = abap_false.

    LOOP AT lt_return ASSIGNING FIELD-SYMBOL(<ls_return>) WHERE type CA 'EA'. "#EC CI_STDSEQ
      IF <ls_return>-id EQ 'CM_EAM_TL_API' AND <ls_return>-number EQ '164' . "Сообщение о блокировке
        cs_pdm_techn-req_eq_pl = '1'.


        MESSAGE w031(zpm_pdm_4129) WITH 'ТК у ЕО ' |{ lv_equnr ALPHA = OUT }| INTO mv_dummy.
        mo_logger->add_syst_msg( is_syst = sy ).

        RETURN.
      ELSE .
        mo_logger->add_bapiret2_msg(
          EXPORTING
            is_bapiret2      =  <ls_return>      ).

        IF sy-subrc <> 0.
*           MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
*             WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.

      mo_logger->add_syst_msg( is_syst = sy ).
      RETURN.
    ENDIF.

    mo_dao->bapi_commit(  ).

    cs_pdm_techn-eq_plnnr  = lv_plnnr.
    cs_pdm_techn-eq_plnal  = lv_plnal.
    cs_pdm_techn-req_eq_pl = '2'.
